#command to list models

aws bedrock list-inference-profiles \
  --region eu-west-2 \
  --type-equals SYSTEM_DEFINED \
  --query "inferenceProfileSummaries[?contains(inferenceProfileId, 'claude')].{ID:inferenceProfileId, ARN:inferenceProfileArn}" \
  --output table



#Add the required policy statements for the Function URL
aws lambda add-permission \
  --function-name cloudboosta-chat-1 \
  --region eu-west-2 \
  --statement-id FunctionURLAllowPublicAccess \
  --action lambda:InvokeFunctionUrl \
  --principal "*" \
  --function-url-auth-type NONE

aws lambda add-permission \
  --function-name cloudboosta-chat-1 \
  --region eu-west-2 \
  --statement-id FunctionURLAllowInvokeAction \
  --action lambda:InvokeFunction \
  --principal "*" \
  --invoked-via-function-url
